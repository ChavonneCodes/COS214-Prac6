66 const std::string& /*outcome*/) override {
67 std::vector<const WorkflowComponent*> result;
68 for (const TransitionDefinition& transition : definition.getTransitions()) {
69 if (transition.getFrom() != completed.getId()) continue;
70 if (const WorkflowComponent* target = findNode(definition, transition.getTo())) result.push_back(target);

124inline WorkflowBlueprint makeBlueprint(const std::string& id, const std::string& name, int version,
125 const std::vector<Step>& steps,
126 const std::vector<std::pair<std::string, std::string>>& links) {
127 WorkflowBlueprint blueprint;
128 blueprint.id = id;
129 blueprint.name = name;
130 blueprint.version = version;
131 blueprint.hasRootTree = true;
132 blueprint.rootTreeData = makeNode(NodeType::STAGE, "root", name);
133
134 std::set<std::string> roles;
135 std::string openStage;
136 for (const Step& step : steps) {
137 WorkflowNodeData leaf = makeNode(NodeType::ACTIVITY, step.id, step.name, step.role);
138 if (step.calls.empty()) {
139 if (roles.insert(step.role).second) blueprint.roles.push_back(RoleDefinition(step.role, step.role, 1));
140 blueprint.activities.push_back(ActivityDefinition(step.id, step.name, step.role));
141 } else {
142 leaf = makeNode(NodeType::SUBPROCESS, step.id, step.name);
143 leaf.calledDefId = step.calls;
144 }
145 std::vector<WorkflowNodeData>& top = blueprint.rootTreeData.children;
146 if (step.stage.empty()) {
147 top.push_back(leaf);
148 } else {
149 if (step.stage != openStage) {
150 std::string stageId = step.stage;
151 stageId.erase(std::remove(stageId.begin(), stageId.end(), ' '), stageId.end());
152 top.push_back(makeNode(NodeType::STAGE, stageId, step.stage));
153 }
154 top.back().children.push_back(leaf);
155 }
156 openStage = step.stage;
157 }
158 for (const auto& link : links) blueprint.transitions.push_back(TransitionDefinition(link.first, link.second, "always"));
159 return blueprint;
160}

169inline std::vector<std::unique_ptr<ApprovalHandler>> buildApprovalChain(const ApprovalChainDefinition& definition,
170 WorkItem* timeoutItem, WorkflowEngine* engine) {
171 std::vector<std::unique_ptr<ApprovalHandler>> handlers;
172 for (const ApprovalStepDefinition& step : definition.getSteps()) {
173 handlers.emplace_back(new RoleApprovalHandler(step.getRoleId()));
174 }
175 if (timeoutItem) handlers.emplace_back(new TimeoutEscalationHandler(timeoutItem, engine));
176 for (size_t i = 0; i + 1 < handlers.size(); ++i) handlers[i]->setNext(handlers[i + 1].get());
177 return handlers;
178}

278 void run(Scenario& scenario) {
279 std::cout << "\n=== " << scenario.title << " ===\n";
280 instanceId.clear();
281 if (scenario.define) scenario.define();
282 const WorkflowDefinition* definition = repository.load(scenario.definitionId, scenario.version);
283 trace.beginCase(scenario.number, scenario.title, *definition, scenario.branches,
284 scenario.calledDefinition.empty() ? nullptr : repository.load(scenario.calledDefinition, 1));
285 beat("define", false, "", "Definition", scenario.definitionId + " v" + std::to_string(scenario.version) +
286 " stored in the repository", "Builder + Composite", "repository.load(id, version)");
287
288 engine.setTreeWrapper(scenario.treeWrapper);
289 instanceId = engine.startWorkflow(scenario.definitionId, scenario.version);
290 WorkflowInstance* instance = engine.getInstance(instanceId);
291 instance->getData().setValue(std::to_string(scenario.budget));
292 bus.notify(WorkFlowEvent("INSTANCE_STARTED", instanceId, "-", "RUNNING"));
293 beat("start", false, "", "Case started", instanceId + " pinned to v" + std::to_string(scenario.version) +
294 ", least-busy person assigned to each step", "Factory Method + Strategy + Decorator",
295 "engine.startWorkflow(id, version)");
296 if (scenario.onStart) scenario.onStart(instanceId);
297
298 std::string currentId = scenario.firstStep;
299 while (!currentId.empty()) {
300 const WorkflowComponent* node = findNode(*definition, currentId);
301 std::string outcome = "approved";
302 if (auto* call = dynamic_cast<const SubProcessDefinition*>(node)) {
303 runSubProcess(*call);
304 } else {
305 WorkItem* item = instance->findWorkItem(currentId);
306 // A Completed item has no way forward, so a routing loop rolls the commands back instead
307 while (item->getState() == "Completed" || item->getState() == "In_Progress") engine.undoLast();
308 auto step = scenario.special.find(currentId);
309 outcome = step != scenario.special.end() ? step->second(item) : startAndComplete(item);
310 }
311
312 auto router = scenario.routers.find(currentId);
313 RoutingStrategy* routing = router != scenario.routers.end() ? router->second : &sequential;
314 std::vector<const WorkflowComponent*> next =
315 routing->next(*node, *definition, outcome);
316 currentId = next.empty() ? "" : next[0]->getId();
317 }
318
319 instance->setState("COMPLETED");
320 bus.notify(WorkFlowEvent("INSTANCE_COMPLETED", instanceId, "-", "COMPLETED"));
321 report(scenario.title);
322 trace.endCase();
323 }

327 void runSubProcess(const SubProcessDefinition& call) {
328 engine.advance(instanceId);
329 beat("run", true, call.getId(), "Sub-process", "The parent pauses while " + call.getCalledDefinitionId() +
330 " runs as its own workflow", "Composite", "engine.advance(instanceId)");
331 WorkflowInstance* child = engine.getInstance(instanceId + "-" + call.getId());
332 for (WorkItem* item : child->getWorkItems()) startAndComplete(item);
333 child->setState("COMPLETED");
334 engine.advance(instanceId);
335 beat("run", true, call.getId(), "Parent resumes", "The child workflow finished, so the parent carries on",
336 "Composite", "engine.advance(instanceId)");
337 }

340 void report(const std::string& title) {
341 std::unique_ptr<RuntimeComponentIterator> walk(
342 engine.getInstance(instanceId)->getRoot()->createIterator(new DFSTraversalStrategy<RuntimeComponent>()));
343 std::string order;
344 while (walk->hasNext()) {
345 order += (order.empty() ? "" : " ") + dynamic_cast<const RuntimeComponent&>(walk->getNext()).getId();
346 }
347 beat("report", false, "", "Iterator walk", order, "Iterator", "root->createIterator(strategy)");
348
349 ProgressReportVisitor progress;
350 WorkloadVisitor workload;
351 engine.applyVisitor(instanceId, progress);
352 engine.applyVisitor(instanceId, workload);
353 beat("report", false, "", "Report", std::to_string(progress.getCompletedItems()) + " of " +
354 std::to_string(progress.getTotalItems()) + " steps done, " +
355 std::to_string(engine.getHistory(instanceId).size()) + " commands in history, " +
356 std::to_string(engine.getAuditTrail(instanceId).size()) + " audit lines",
357 "Visitor + Observer", "engine.applyVisitor(id, visitor), engine.getAuditTrail(id)");
358 summary.push_back(title + ": " + std::to_string(progress.getCompletedItems()) + " of " +
359 std::to_string(progress.getTotalItems()) + " steps done, " +
360 std::to_string(engine.getHistory(instanceId).size()) + " commands");
361 }
