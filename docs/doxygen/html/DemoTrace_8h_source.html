47 static std::string quote(const std::string& text) {
48 std::string out = "\"";
49 for (char c : text) {
50 if (c == '"' || c == '\\') out += std::string("\\") + c;
51 else if (c == '\n') out += "\\n";
52 else out += c;
53 }
54 return out + "\"";
55 }

58 static std::string quoteAll(const std::vector<std::string>& lines) {
59 std::string out = "[";
60 for (size_t i = 0; i < lines.size(); ++i) out += (i ? "," : "") + quote(lines[i]);
61 return out + "]";

79 void beginCase(int scenario, const std::string& title, const WorkflowDefinition& definition,
80 const std::vector<Branch>& branches, const WorkflowDefinition* called = nullptr) {
81 std::string nodes = "[", edges = "[";
82 std::vector<ActivityDefinition> activities = definition.getActivities();
